/* HyperFamily companion bridge — injected into every served page.
 *
 * The phone's WebView loads this application from the workstation over HTTP,
 * where no Electron preload exists. This bridge provides window.hyperfamily
 * on top of the companion /api endpoints, so the renderer cannot tell the
 * difference for everything the phone supports. Modules the companion does
 * not implement answer with a clear "not available" error instead of failing
 * silently. The token arrives once in the URL (?companion=…) from the mobile
 * shell and is kept in this origin's localStorage from then on. */
;(function () {
  var KEY = 'hyperfamily.companion.token'
  try {
    var params = new URLSearchParams(window.location.search)
    var incoming = params.get('companion')
    if (incoming) {
      window.localStorage.setItem(KEY, incoming)
      window.history.replaceState(null, '', window.location.pathname)
    }
  } catch (e) {
    /* private modes without localStorage simply re-send the token each time */
  }

  function token() {
    try {
      return window.localStorage.getItem(KEY) || ''
    } catch (e) {
      return ''
    }
  }

  function call(method, url, body) {
    return window
      .fetch(url, {
        method: method,
        headers: { 'content-type': 'application/json', authorization: 'Bearer ' + token() },
        body: body === undefined ? undefined : JSON.stringify(body)
      })
      .then(function (response) {
        return response.text().then(function (text) {
          var data = null
          try {
            data = text ? JSON.parse(text) : null
          } catch (e) {
            data = null
          }
          if (!response.ok) throw new Error((data && data.error) || 'Companion request failed (' + response.status + ')')
          return data
        })
      })
  }

  function unavailable(feature) {
    return function () {
      return Promise.reject(new Error(feature + ' is not available in the companion app'))
    }
  }

  var SUBSCRIBERS = {
    subscribe: 1,
    onStatus: 1,
    onStep: 1,
    onProgress: 1,
    onVersion: 1,
    onAgentStep: 1,
    onInstallStep: 1,
    onFinished: 1,
    onInstallFinished: 1,
    onDataChange: 1
  }
  function silentSubscription() {
    return function () {}
  }

  // Live terminal (v3.10.0): the workstation streams each session's output
  // into a ring buffer; this poller drains it and fans out to the xterm
  // emulator exactly like the desktop's IPC events do.
  var terminalListeners = { data: new Set(), status: new Set() }
  var terminalTimer = null
  var terminalSeq = 0

  function pollTerminal() {
    call('GET', '/api/terminal/events?after=' + terminalSeq)
      .then(function (events) {
        ;(events || []).forEach(function (event) {
          terminalSeq = Math.max(terminalSeq, event.seq || 0)
          var set = event.channel === 'terminal:data' ? terminalListeners.data : terminalListeners.status
          set.forEach(function (listener) {
            try {
              listener(event.payload)
            } catch (e) {}
          })
        })
      })
      .catch(function () {})
  }

  function terminalSubscribe(kind) {
    return function (callback) {
      terminalListeners[kind].add(callback)
      if (!terminalTimer) terminalTimer = window.setInterval(pollTerminal, 700)
      pollTerminal()
      return function () {
        terminalListeners[kind].delete(callback)
        if (!terminalListeners.data.size && !terminalListeners.status.size && terminalTimer) {
          window.clearInterval(terminalTimer)
          terminalTimer = null
        }
      }
    }
  }

  // Store-update events (v3.10.2): version, step, progress, agentStep, installStep, finished, installFinished
  var storeUpdateListeners = {
    version: new Set(),
    step: new Set(),
    progress: new Set(),
    agentStep: new Set(),
    installStep: new Set(),
    finished: new Set(),
    installFinished: new Set()
  }
  var storeUpdateTimer = null
  var storeUpdateSeq = 0

  var STORE_CHANNEL_MAP = {
    'store-update:version': 'version',
    'store-update:step': 'step',
    'store-update:progress': 'progress',
    'store-update:agent-step': 'agentStep',
    'store-update:install-step': 'installStep',
    'store-update:finished': 'finished',
    'store-update:install-finished': 'installFinished'
  }

  function pollStoreUpdate() {
    call('GET', '/api/store-update/events?after=' + storeUpdateSeq)
      .then(function (events) {
        ;(events || []).forEach(function (event) {
          storeUpdateSeq = Math.max(storeUpdateSeq, event.seq || 0)
          var kind = STORE_CHANNEL_MAP[event.channel]
          if (!kind) return
          var set = storeUpdateListeners[kind]
          if (!set) return
          set.forEach(function (listener) {
            try {
              listener(event.payload)
            } catch (e) {}
          })
        })
      })
      .catch(function () {})
  }

  function storeUpdateSubscribe(kind) {
    return function (callback) {
      var set = storeUpdateListeners[kind]
      if (!set) return silentSubscription()()
      set.add(callback)
      if (!storeUpdateTimer) storeUpdateTimer = window.setInterval(pollStoreUpdate, 600)
      pollStoreUpdate()
      return function () {
        set.delete(callback)
        var any = false
        for (var k in storeUpdateListeners) {
          if (storeUpdateListeners[k].size) {
            any = true
            break
          }
        }
        if (!any && storeUpdateTimer) {
          window.clearInterval(storeUpdateTimer)
          storeUpdateTimer = null
        }
      }
    }
  }

  var monitorListeners = new Set()
  var monitorTimer = null

  function pollMonitor() {
    call('GET', '/api/monitor')
      .then(function (snapshot) {
        monitorListeners.forEach(function (listener) {
          try {
            listener(snapshot)
          } catch (e) {}
        })
      })
      .catch(function () {})
  }

  var api = {
    app: {
      info: function () {
        return call('GET', '/api/app/info')
      },
      openExternal: function (url) {
        window.open(url, '_blank')
        return Promise.resolve()
      },
      listServerFiles: function (dir) {
        var q = dir ? '?dir=' + encodeURIComponent(dir) : ''
        return call('GET', '/api/app/files' + q)
      }
    },
    auth: {
      status: function () {
        return Promise.resolve({ authenticated: true })
      },
      login: function (form) {
        return call('POST', '/api/auth/login', form)
      },
      logout: function () {
        return Promise.resolve({ success: true })
      },
      changePassword: unavailable('Changing credentials'),
      updateCredentials: unavailable('Changing credentials'),
      recoverStatus: unavailable('PIN recovery'),
      recover: unavailable('PIN recovery')
    },
    monitor: {
      snapshot: function () {
        return call('GET', '/api/monitor')
      },
      subscribe: function (callback) {
        monitorListeners.add(callback)
        if (!monitorTimer) monitorTimer = window.setInterval(pollMonitor, 5000)
        pollMonitor()
        return function () {
          monitorListeners.delete(callback)
          if (!monitorListeners.size && monitorTimer) {
            window.clearInterval(monitorTimer)
            monitorTimer = null
          }
        }
      }
    },
    branches: {
      list: function () {
        return call('GET', '/api/branches')
      },
      save: function (payload) {
        return call('POST', '/api/branches', payload)
      },
      remove: function (id) {
        return call('POST', '/api/branches/remove', { id: id })
      },
      removeAll: unavailable('Wiping the directory')
    },
    devices: {
      list: function () {
        return call('GET', '/api/devices')
      },
      save: function (payload) {
        return call('POST', '/api/devices', payload)
      },
      remove: function (id) {
        return call('POST', '/api/devices/remove', { id: id })
      }
    },
    settings: {
      get: function () {
        return call('GET', '/api/settings')
      },
      save: function (patch) {
        return call('POST', '/api/settings', patch)
      }
    },
    notes: {
      list: function () {
        return call('GET', '/api/notes')
      },
      save: function (payload) {
        return call('POST', '/api/notes', payload)
      },
      remove: function (id) {
        return call('POST', '/api/notes/remove', { id: id })
      }
    },
    snippets: {
      list: function () {
        return call('GET', '/api/snippets')
      },
      save: function (payload) {
        return call('POST', '/api/snippets', payload)
      },
      remove: function (id) {
        return call('POST', '/api/snippets/remove', { id: id })
      }
    },
    inventory: {
      list: function () {
        return call('GET', '/api/inventory')
      }
    },
    credentials: {
      list: function () {
        return call('GET', '/api/credentials')
      },
      map: function () {
        return call('GET', '/api/credentials/map')
      },
      overview: function () {
        return call('GET', '/api/credentials/overview')
      }
    },
    terminal: {
      targets: function () {
        return call('GET', '/api/terminal/targets')
      },
      open: function (payload) {
        return call('POST', '/api/terminal/open', payload)
      },
      write: function (payload) {
        return call('POST', '/api/terminal/write', payload)
      },
      resize: function (payload) {
        return call('POST', '/api/terminal/resize', payload).catch(function () {})
      },
      close: function (sessionId) {
        return call('POST', '/api/terminal/close', { sessionId: sessionId })
      },
      onData: terminalSubscribe('data'),
      onStatus: terminalSubscribe('status')
    },
    storeUpdate: {
      version: function (payload) {
        return call('POST', '/api/store-update/version', payload)
      },
      versionCache: function () {
        return call('GET', '/api/store-update/version-cache')
      },
      versions: function (payload) {
        return call('POST', '/api/store-update/versions', payload)
      },
      installed: function (payload) {
        return call('POST', '/api/store-update/installed', payload)
      },
      importAgent: function (payload) {
        return call('POST', '/api/store-update/import-agent', payload)
      },
      importAgentAll: function (payload) {
        return call('POST', '/api/store-update/import-agent-all', payload)
      },
      cancelAgentImport: function (payload) {
        return call('POST', '/api/store-update/cancel-agent-import', payload)
      },
      deploy: function (payload) {
        return call('POST', '/api/store-update/deploy', payload)
      },
      deployAll: function (payload) {
        return call('POST', '/api/store-update/deploy-all', payload)
      },
      testAccess: function (payload) {
        return call('POST', '/api/store-update/test-access', payload)
      },
      install: function (payload) {
        return call('POST', '/api/store-update/install', payload)
      },
      installAll: function (payload) {
        return call('POST', '/api/store-update/install-all', payload)
      },
      cancelInstall: function (payload) {
        return call('POST', '/api/store-update/cancel-install', payload)
      },
      onVersion: storeUpdateSubscribe('version'),
      onStep: storeUpdateSubscribe('step'),
      onProgress: storeUpdateSubscribe('progress'),
      onAgentStep: storeUpdateSubscribe('agentStep'),
      onInstallStep: storeUpdateSubscribe('installStep'),
      onFinished: storeUpdateSubscribe('finished'),
      onInstallFinished: storeUpdateSubscribe('installFinished')
    }
  }

  window.hyperfamily = new Proxy(api, {
    get: function (target, property) {
      if (property in target) return target[property]
      if (typeof property !== 'string') return undefined
      return new Proxy(
        {},
        {
          get: function (_target, method) {
            var name = String(method)
            if (SUBSCRIBERS[name]) return silentSubscription
            if (name === 'palette')
              return function () {
                return Promise.resolve(null)
              }
            return unavailable(property + '.' + name)
          }
        }
      )
    }
  })
})()
